```
    #加载默认配置
make ARCH=arm64 rockchip_linux_defconfig

    #修改保存到.config后要覆盖保存回原来的defconfig文件
make ARCH=arm64 savedefconfig
    #用新生成的defconfig覆盖原来的默认配置
cp defconfig arch/arm64/configs/rockchip_linux_defconfig
```

```
引脚编号：gpio3 RK_PC1
gpio0 gpio1 gpio2 每组32个
A0~A7 B0~B7 每组8个引脚
3*32+17 = 113


cat /sys/kernel/debug/gpio

# 查看系统识别到的所有存储设备
fdisk -l

查看RAM
free -h  # 以人类可读的单位（如 GB、MB）显示
cat /proc/meminfo  #可查看更详细的信息，如 MemTotal（总内存）、MemFree（空闲内存）、Buffers（缓存的文件元数据）、Cached（缓存的文件内容）等

查看磁盘
fdisk -l

查看已识别到的硬盘容量
df-h



控制引脚
cat /sys/kernel/debug/gpio
# 导出
echo 97 > /sys/class/gpio/export
# 查看是否存在目录
ls /sys/class/gpio/gpio97
# 设置为输出
echo out > /sys/class/gpio/gpio97/direction
# 拉低电平（复位SSD）
echo 0 > /sys/class/gpio/gpio97/value
# 拉高电平（释放复位）
echo 1 > /sys/class/gpio/gpio97/value
#清理
echo 97 > /sys/class/gpio/unexport

echo 97 > /sys/class/gpio/export
echo out > /sys/class/gpio/gpio97/direction
echo 0 > /sys/class/gpio/gpio97/value
sleep 0.1
echo 1 > /sys/class/gpio/gpio97/value
sleep 0.1
echo 1 > /sys/bus/pci/rescan

echo 113 > /sys/class/gpio/export
echo out > /sys/class/gpio/gpio113/direction
echo 0 > /sys/class/gpio/gpio113/value
echo 1 > /sys/class/gpio/gpio113/value

cat /sys/class/gpio/gpio113/value
```
