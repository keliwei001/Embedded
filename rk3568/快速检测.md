###### 网络

```
dmesg | grep -i "gmac\|eth\|stmmac"
dmesg | tail -20
dmesg | grep -i phy
dmesg | grep -i "phy.*driver"
ifconfig -a
ip link show
ip link set eth0 up
ethtool eth0

udhcpc -i eth0
ip addr add 192.168.1.100/24 dev eth0
ip route add default via 192.168.1.1
ip addr show eth0
ping 192.168.1.1       # Ping你的路由器
ping 8.8.8.8           # Ping Google DNS
```
