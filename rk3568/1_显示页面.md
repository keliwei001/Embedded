###### 查看连接的显示器

```
# 查看 DRM 设备信息
cat /sys/kernel/debug/dri/0/state

# 查看所有连接的显示设备
ls /sys/class/drm/

# 查看每个接口的状态
cat /sys/class/drm/card0-*/status

# 运行一个全屏测试程序来确认显示参数
weston-simple-egl

# 查看 DRM 连接器状态
find /sys/class/drm -name "status" -exec echo {} \; -exec cat {} \;

# 或者一行命令查看所有显示接口状态
for connector in /sys/class/drm/card*-*/status; do echo "$(basename $(dirname $connector)): $(cat $connector)"; done
```
